// ECHO TUNNEL — the scene is JUST a spectral ring and a spark of core.
// Tunnel Push re-projects each frame outward: the ring's own past
// becomes an endless corridor. Swirl bends it; kick punches it deeper.
vec3 scene(vec2 p, float t){
  float r = length(p);
  float a = atan(p.y, p.x);
  float ring = exp(-abs(r - 0.28)*22.0);
  vec3 col = spectral(a/6.28318 + t*0.05 + uHue) * ring * 1.6;
  col += vec3(1.0, 0.95, 0.85) * exp(-r*14.0) * 0.8;
  return col;
}

// LIQUID SMEAR — three drifting points of colored light. Persistence
// keeps their past alive; Liquid warps the ghosts into oil. This is
// the light-show smear the paint field hints at, running free.
vec3 scene(vec2 p, float t){
  vec3 col = vec3(0.0);
  for(int i = 0; i < 3; i++){
    float fi = float(i);
    vec2 c = vec2(sin(t*(0.4 + fi*0.13) + fi*2.1), cos(t*(0.31 + fi*0.11) + fi*4.2))*0.5;
    float d = length(p - c);
    col += spectral(fi*0.33 + t*0.03 + uHue) * exp(-d*d*90.0) * 1.8;
  }
  return col;
}

  scene:`
// EMBER RAIN — sparse embers rise on a grid of hashes. With
// Persistence high their motion writes ribbons; Liquid makes the
// ribbons waver like heat. Try Ghost Hue ~.3 for spectral smoke.
vec3 scene(vec2 p, float t){
  vec2 g = vec2(p.x*6.0, p.y*3.0 - t*0.9*uFlow);
  vec2 id = floor(g);
  vec2 f = fract(g) - 0.5;
  float h = hash(id);
  vec2 off = (vec2(hash(id + 1.7), hash(id + 3.1)) - 0.5)*0.6;
  float d = length(f - off);
  float on = step(0.75, h);
  vec3 col = mix(vec3(1.0, 0.45, 0.1), vec3(1.0, 0.85, 0.3), hash(id + 5.0)) * exp(-d*d*180.0) * on * 2.0;
  return col;
}
